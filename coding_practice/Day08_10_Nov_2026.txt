  hosts: all
  become: yes
  tasks:
    - name: Install dependencies
      yum:
        name: "dark3-pip"
        state: present
    - name: Install docker-compose
      pip:
        name: docker-compose

resource "aws_security_group" "allow_ssh_http" {
  name        = "allow_ssh_http"
  description = "Allow SSH and HTTP access"
  ingress {
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  ingress {
    from_port   = 80
    to_port     = 80
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }
}
resource "aws_instance" "dark_ec2" {
  ami           = "ami-0abcd1234abcd1234"
  instance_type = "t3.medium"
  security_groups = [aws_security_group.allow_ssh_http.name]

  ebs_block_device {
    device_name = "/dev/sdh"
    volume_size = 30
  }
}
apiVersion: v1
kind: Service
metadata:
  name: dark-service
spec:
  type: NodePort
  selector:
    app: dark-app
  ports:
    - protocol: TCP
      port: 80
      nodePort: 30007
pipeline {
    agent any
    stages {
        stage('Build Docker Image') {
            steps {
                script {
                    docker.build('dark-app')
                }
            }
        }
        stage('Push Docker Image') {
            steps {
                script {
                    docker.withRegistry('https://docker.io', 'docker-credentials') {
                        docker.image('dark-app').push('latest')
                    }
                }
            }
        }
    }
}
stages:
  - build
  - push

build:
  stage: build
  script:
    - docker build -t dark-app .

push:
  stage: push
  script:
    - docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
    - docker push dark-app
